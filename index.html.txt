    <script>
        const inputs = document.querySelectorAll('input, select, textarea');
        const noteDisplay = document.getElementById('clinicalNote');

        // 设置默认为今天的日期
        document.getElementById('visitDate').valueAsDate = new Date();

        function generateNote() {
            // 获取表单数据
            const visitDate = document.getElementById('visitDate').value;
            const formattedDate = visitDate ? new Date(visitDate).toLocaleDateString('en-GB') : '';
            
            const weight = document.getElementById('weight').value || '-';
            const diagnosis = document.getElementById('diagnosis').value;
            const transfusionType = document.getElementById('transfusionType').value;
            
            const ferritin = document.getElementById('ferritin').value || '-';
            const mriCardiac = document.getElementById('mriCardiac').value || '-';
            const mriLiver = document.getElementById('mriLiver').value || '-';
            
            const deferiproneDose = document.getElementById('deferiproneDose').value || 'Nil';
            const deferasiroxDose = document.getElementById('deferasiroxDose').value || 'Nil';
            const compliance = document.getElementById('compliance').value || 'Nil';
            
            const otherIssues = document.getElementById('otherIssues').value || 'Nil';
            const plan = document.getElementById('plan').value || 'Nil';

            // 格式化输出文本 (与原版 CSV TMTAC 格式对齐)
            const noteText = `=== TMTAC PHARMACIST NOTES ===
Date: ${formattedDate}
Weight (kg): ${weight}
Diagnosis: ${diagnosis}
Transfusional Types: ${transfusionType}

[ LAB INVESTIGATIONS ]
Serum Ferritin (ng/mL): ${ferritin}
MRI T2* (Cardiac): ${mriCardiac}
MRI T2* (Liver): ${mriLiver}

[ OTHER ISSUES ]
${otherIssues}

[ CHELATION THERAPY ]
Deferiprone Dose: ${deferiproneDose}
Deferasirox Dose: ${deferasiroxDose}
Compliance / Missed Doses: ${compliance}

[ PHARMACIST PLAN ]
${plan}`;

            noteDisplay.textContent = noteText;
        }

        // 监听所有输入框，有变动时立即更新右侧文本
        inputs.forEach(input => {
            input.addEventListener('input', generateNote);
            input.addEventListener('change', generateNote);
        });

        // 复制到剪贴板功能
        function copyNote() {
            const textToCopy = noteDisplay.textContent;
            navigator.clipboard.writeText(textToCopy).then(() => {
                const btn = document.querySelector('.copy-btn');
                const originalText = btn.textContent;
                btn.textContent = "✓ Copied Successfully!";
                btn.style.background = "#059669";
                setTimeout(() => { 
                    btn.textContent = originalText; 
                    btn.style.background = "#10b981";
                }, 2000);
            }).catch(err => {
                console.error('Failed to copy: ', err);
                alert("Failed to copy. Please manually copy the text.");
            });
        }

        // 初始化第一次渲染
        generateNote();
    </script>
